  attack:
  - T1564.001

  comments:
  - detection rule - identify creation of hidden (dot-prefixed) files/dirs in /tmp,
    /dev/shm, /var/tmp (attacker staging)
match-on:
  events:
    kunai:
    - file_create
matches:
  $hidden_tmp: .data.path ~= "^/tmp/\.[^/]+"
  $hidden_shm: .data.path ~= "^/dev/shm/\.[^/]+"
  $hidden_vtmp: .data.path ~= "^/var/tmp/\.[^/]+"
condition: $hidden_tmp or $hidden_shm or $hidden_vtmp
severity: 7
